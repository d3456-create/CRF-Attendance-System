📋 How to connect Google Sheets (one-time setup):

  1. Open Google Sheets → create a new sheet and name it Attendance.
  2. Go to Extensions → Apps Script, paste the Apps Script code below, and click Save.
  3. Click Deploy → New Deployment → Web App. Set "Who has access" to Anyone. Copy the URL.
  4. Paste the URL above and click Save URL.
  5. On first submit, Google will ask you to authorize — click Allow.

Apps Script Code — paste this in your Google Apps Script editor:

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    // ── Sheet 1: Session Summary ──────────────────────────────────
    var sheet1 = ss.getSheetByName("Session Summary") || ss.insertSheet("Session Summary");
    if (sheet1.getLastRow() === 0) {
      sheet1.appendRow(["#","Faculty","Subject","Class/Batch","Date","Time",
                        "Present","Absent","Leave","Total","Attendance %"]);
    }

    // ── Sheet 2: Student Detail ───────────────────────────────────
    var sheet2 = ss.getSheetByName("Student Detail") || ss.insertSheet("Student Detail");
    if (sheet2.getLastRow() === 0) {
      sheet2.appendRow(["Session #","Roll No","Student Name","Faculty",
                        "Subject","Class/Batch","Date","Status"]);
    }

    var data = JSON.parse(e.postData.contents);
    var d    = new Date(data.date);
    var dateStr = Utilities.formatDate(d, Session.getScriptTimeZone(), "dd/MM/yyyy");
    var timeStr = Utilities.formatDate(d, Session.getScriptTimeZone(), "HH:mm");
    var sessionId = "#" + String(data.id).padStart(3,"0");

    sheet1.appendRow([sessionId, data.faculty, data.subject, data.cls,
                      dateStr, timeStr, data.present, data.absent,
                      data.leave, data.total, data.pct + "%"]);

    data.studentData.forEach(function(sd) {
      sheet2.appendRow([sessionId, sd.roll, sd.name, data.faculty,
                        data.subject, data.cls, dateStr,
                        sd.status === "P" ? "Present" : sd.status === "A" ? "Absent" : "Leave"]);
    });

    return ContentService
      .createTextOutput(JSON.stringify({status:"ok"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({status:"error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
